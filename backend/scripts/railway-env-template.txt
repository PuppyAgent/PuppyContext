# ============================================================
# Railway 环境变量模板

# 1. 复制此文件内容

# ==================== 通用配置 (所有服务) ====================

# Supabase 数据库 (必需；后端需要 service_role key)
SUPABASE_URL=https://your-project.supabase.co
SUPABASE_KEY=your-supabase-service-role-key

# S3 存储 (必需)
S3_ENDPOINT_URL=https://s3.amazonaws.com
S3_BUCKET_NAME=your-bucket-name
S3_REGION=us-east-1
S3_ACCESS_KEY_ID=your-access-key
S3_SECRET_ACCESS_KEY=your-secret-key

# JWT 安全配置 (必需)
JWT_SECRET=your-secure-random-secret-key
JWT_ALGORITHM=HS256

# Redis (必需 - 从 Railway Redis 获取 Internal URL)
ETL_REDIS_URL=redis://default:xxx@redis.railway.internal:6379
AUTH_SECURITY_REDIS_URL=redis://default:xxx@redis.railway.internal:6379
NOTIFICATIONS_REDIS_URL=redis://default:xxx@redis.railway.internal:6379

# 应用配置
APP_ENV=production
DEBUG=False
APP_NAME=ContextBase
VERSION=1.0.0
ALLOWED_HOSTS=https://your-frontend.com
PUBLIC_URL=https://your-api.railway.app
FRONTEND_URL=https://your-frontend.com
SUPABASE_PUBLIC_URL=https://your-project.supabase.co
DESKTOP_AUTH_PUBLIC_BASE_URL=https://your-api.railway.app/api/v1
INTERNAL_API_SECRET=your-shared-internal-secret
ACCESS_CREDENTIAL_HASH_SECRET=your-stable-random-access-credential-hmac-secret
ENTITLEMENTS_MODE=db
BILLING_ENFORCEMENT=required
